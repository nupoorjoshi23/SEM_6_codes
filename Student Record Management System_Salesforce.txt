Student Record Management System

Step 1: Create the Database Table (Custom Object)
Click the Gear Icon (top right) and select Setup.
Navigate to Object Manager -> Create -> Custom Object.
Set the Label to Student and Plural Label to Students. Save it. (Salesforce automatically creates the API name Student__c).

Step 2: Create the Database Columns (Custom Fields)
While still in the Student object menu, click Fields & Relationships.
Click New to create your four custom fields:
Select Text -> Next -> Label: Name -> Next -> Save. (API Name: Name__c)
Select Number -> Next -> Label: Roll_No -> Next -> Save. (API Name: Roll_No__c)
Select Text -> Next -> Label: Class -> Next -> Save. (API Name: Class__c)
Select Phone -> Next -> Label: Mobile No -> Next -> Save. (API Name: Mobile_No__c)

Step 3: Write the Backend Logic (Apex Class)
Click the Gear Icon -> Developer Console.
Go to File -> New -> Apex Class.
Name it StudentManagementController.
Paste the Apex code and save (Ctrl+S).

public class StudentManagementController {
    public Student__c student {get;set;}
    public List<Student__c> studentList {get;set;}
    
    public StudentManagementController(){
		student = new Student__c();
        loadStudents();
    }
    //READ
    public void loadStudents(){
		studentList = [
            SELECT Id, Name__c, Roll_No__c, Class__c, Mobile_No__c
            FROM Student__c
            ORDER BY Roll_No__c
        ];
    }
    //CREATE & UPDATE
    public void saveStudent(){
        upsert student;
        student = new Student__c(); //clear the previous data and create new
        loadStudents();
    }
    //PREPARE FOR UPDATE
    public void editStudent(){
		String sid = ApexPages.currentPage().getParameters().get('sid');
        if(sid != null){
            // Fetch the selected student and put their data back into the top form
			student = [
                SELECT Name__c, Roll_No__c, Class__c, Mobile_No__c
            	FROM Student__c
            	WHERE Id = :sid
                LIMIT 1
            ];
        }
    }
    //DELETE
    public void deleteStudent(){
		String sid = ApexPages.currentPage().getParameters().get('sid');
        if(sid != null){
			Student__c s = [SELECT Id FROM Student__c WHERE Id = :sid LIMIT 1];
            delete s;
            loadStudents();
        }
    }
}

Step 4: Build the Frontend (Visualforce Page)
In the Developer Console, go to File -> New -> Visualforce Page.
Name it StudentManagementPage.
Paste the Visualforce markup provided and save.

<apex:page controller="StudentManagementController">
    <apex:form id="frm">
        
        <apex:pageBlock title="Student Form">
        	<apex:pageBlockSection columns="1">
                <apex:inputField value="{!student.Name__c}"/>
                <apex:inputField value="{!student.Roll_No__c}"/>
                <apex:inputField value="{!student.Class__c}"/>
                <apex:inputField value="{!student.Mobile_No__c}"/>
                
                <apex:commandButton value="Save Student" action="{!saveStudent}" rerender="frm"/>
            </apex:pageBlockSection>
        </apex:pageBlock>
        
        <apex:pageBlock title="Student List">
        	<apex:pageBlockTable value="{!studentList}" var="s">
            	<apex:column value="{!s.Name__c}"/>
                <apex:column value="{!s.Roll_No__c}"/>
                <apex:column value="{!s.Class__c}"/>
                <apex:column value="{!s.Mobile_No__c}"/>
                
                <apex:column headerValue="Actions">
                    <apex:commandButton value="Edit" action="{!editStudent}" rerender="frm">
                        <apex:param name="sid" value="{!s.Id}"/>
                    </apex:commandButton>
                    <apex:commandButton value="Delete" action="{!deleteStudent}" rerender="frm">
                        <apex:param name="sid" value="{!s.Id}"/>
                    </apex:commandButton>
                </apex:column>
            </apex:pageBlockTable>	
        </apex:pageBlock>
        
    </apex:form>

</apex:page>
