#include <bits/stdc++.h>
using namespace std;
// tracking arrays (bounds)
vector<int>col_safe; // if col_safe[3]==1, there is a queen in col 3
vector<int>diag1_safe; //normal diagonal
vector<int>diag2_safe; //anti diagonal
vector<vector<int>>board;
int n;

void printBoard(){
    cout << "\nSolution for " << n << "-Queens" << endl;
    for(int i=0;i<n;i++){
        for(int j=0;j<n;j++){
            if(board[i][j]) cout << "Q";
            else cout << ".";
        }
        cout << endl;
    }
}

bool solveNQueens(int row){
    if(row==n) return true; //BASE CASE - all queens are placed from 0 to n-1 row
    //try placing a queen in all columns one by one for the current row
    for(int col=0;col<n;col++){
        //Branch & bound- check in O(1) time
        if(!col_safe[col] && !diag1_safe[row-col+n-1] && !diag2_safe[row+col]){
            board[row][col]=1; //place the queen
            col_safe[col]=1;
            diag1[row-col+n-1]=1; // normal diagonal = r-c+n-1 (add n-1 to avoid negative index)
            diag2_safe[row+col]=1; //anti diagonal = r+c
            
            //recurse to place rest of the queens
            if(solveNQueens(row+1)) return true; // stop seraching once the first solution is found
            //Backtrack if placing queen here, does not give a solution
            board[row][col]=0;
            col_safe[col]=0;
            diag1[row-col+n-1]=0;
            diag2_safe[row+col]=0;
        }
    }
    //if no column is safe in this row, return false to trigger backtracking
    return false;
}

int main() {
    cout << "Enter the number of queens (n): ";
    cin >> n;
    if(n==2 || n==3){
        cout << "No solution exists for " << n << "-Queens problem." << endl;
        return 0;
    }
    board.assign(n, vector<int>(n,0));
    col_safe.assign(n,0);
    diag1_safe.assign(2*n, 0); //to cover all possible diagonal index values
    diag2_safe.assign(2*n, 0);

    if(solveNQueens(0)) printBoard();
    else cout << "No solution exists" << endl;
    return 0;
}
