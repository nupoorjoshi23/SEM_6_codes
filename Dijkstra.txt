#include<bits/stdc++.h>
using namespace std;

typedef pair<int,int> pii; //shortcut- pairs to store {distance,node}

void dijkstra(int start, int v, vector<vector<pii>>& adj){ //pii: {neighbor,wt}
    priority_queue<pii, vector<pii>, greater<pii>>pq; //min-heap using greater
    vector<int>dist(v,INT_MAX); //to store distance of all nodes
    dist[start]=0;
    pq.push({0,start});
    
    while(!pq.empty()){
        int d=pq.top().first; //distance
        int u=pq.top().second; //node
        pq.pop();
        //if d is greater than the best distance we already have skip it
        if(d>dist[u]) continue;
        for(auto& edge:adj[u]){
            int neighbor=edge.first;
            int weight=edge.second;
            if(dist[u]+weight < dist[neighbor]){
                dist[neighbor]=dist[u]+weight;
                pq.push({dist[neighbor], neighbor});
            }
        }
    }
    cout << "---------------Network Routing----------------" << endl;
    for(int i=0;i<v;i++){
        cout << "Latency to Router " << i << ": ";
        if(dist[i]==INT_MAX) cout << "Unreachable" << endl;
        else cout << dist[i] << " ms" << endl;
    }
}
int main() {
    int routers, connections;
    cout << "Enter the number of routers and connections: ";
    cin >> routers >> connections;
    vector<vector<pii>>adj(routers);
    cout << "Enter router1 and router2 and the cost: " << endl;
    for(int i=0;i<connections;i++){
        int u,v,w;
        cin >> u >> v >> w;
        adj[u].push_back({v,w});
        adj[v].push_back({u,w});
    }
    int start;
    cout << "Enter the start node: ";
    cin >> start;
    dijkstra(start, routers, adj);

    return 0;
}
