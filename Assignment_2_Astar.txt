#include<bits/stdc++.h>
using namespace std;

using State = vector<vector<int>>; //3x3 grid
map<int,pair<int,int>>goalCoords; // tile no(1-8) -> {r,c}

struct Node{ // specific moment on time on the board
    State state;
    int g; //cost to reach
    int h; // manhattan distance
    int f; // evaluation function
    string move; // move that resulted in this state(up/down/left/right)
    Node* parent; //pointer to the previous node
    Node(State s, int g_val, string m, Node* p): state(s), g(g_val), move(m), parent(p){
        h=calculateManhattan(s);
        f=g+h;
    }
    int calculateManhattan(const State& s){
        int distance=0;
        for(int r=0;r<3;r++){
            for(int c=0;c<3;c++){
                int tile=s[r][c];
                if(tile!=0){
                    pair<int,int>target=goalCoords[tile];
                    distance += abs(r-target.first)+abs(c-target.second);
                }
            }
        }
        return distance;
    }
    struct CompareNode{ //comparator for priority queue
        bool operator() (const Node* a, const Node* b) const {
            return a->f > b->f; //flips logic of priority queue;turns into minheap
        }
    };
};
pair<int,int> findBlank(const State& s){ // find blank tile coordinate
    for(int r=0;r<3;r++){
        for(int c=0;c<3;c++){
            if(s[r][c]==0) return {r,c};
        }
    }
    return {-1,-1};
}
void printState(const State& s){ //Print the board
    for(const auto& row:s){
        cout << "[ ";
        for(int val:row) cout << val << " ";
        cout << " ]" << endl;
    }
    cout << endl;
}
int countInversions(const State& s){
    vector<int>flat; //squish the 2D grid into 1D line
    for(const auto& row:s){
        for(int val:row){
            if(val!=0) flat.push_back(val); //ignore the blank tile
        }
    }
    int inv_count=0; //An inversion is when a larger number appears before a smaller number.
    for(int i=0;i<flat.size();i++){
        for(int j=i+1;j<flat.size();j++){
            if(flat[i]>flat[j]) inv_count++;
        }
    }
    return inv_count;
}
//If inversion of both start and goal is even, or odd, it is solvable.
bool isSolvable(const State& start, const State& goal){
    int startInv=countInversions(start);
    int goalInv=countInversions(goal);
    return (startInv%2 == goalInv%2);
}
void solvePuzzle(State start, State goal){
    goalCoords.clear(); //initialize gaol coordinates map
    for(int r=0;r<3;r++){
        for(int c=0;c<3;c++){
            if(goal[r][c]!=0) goalCoords[goal[r][c]]={r,c};
        }
    }
    priority_queue<Node*, vector<Node*>, Node::CompareNode>openList;
    set<State>closedList; //to prevent infinite loops
    
    Node* root = new Node(start,0,"Start",NULL);
    openList.push(root);
    cout << "Searching for solution.." << endl;
    while(!openList.empty()){
        Node* current=openList.top();
        openList.pop();
        if(current->state==goal){ //GOAL REACHED
            vector<Node*>path;
            while(current!=NULL){ //backtrack to find the path
                path.push_back(current);
                current=current->parent;
            }
            reverse(path.begin(),path.end()); //path from Start->Goal
            cout << "\nSuccess! Solution found in " << path.size()-1 << " moves\n";
            for(int i=0;i<path.size();i++){
                Node* n=path[i];
                cout << "-----------------------------------------" << endl;
                if(i==0) cout << "Step 0: Initial State" << endl;
                else cout << "Step " << i << ": Move " << n->move << endl;
                cout << "f(n) = " << n->f << " (g(n)=" << n->g << ", h(n)=" << n->h << ")" << endl;
                printState(n->state);
            }
            return;
        }
        //CRITICAL LINE
        if(closedList.count(current->state)) continue;
        closedList.insert(current->state);
        
        pair<int,int>blank=findBlank(current->state);
        int r=blank.first;
        int c=blank.second;
        int dr[]={-1,1,0,0};
        int dc[]={0,0,-1,1}; 
        string moves[]={"Up", "Down", "Left", "Right"};
        for(int i=0;i<4;i++){
            int nr=r+dr[i];
            int nc=c+dc[i];
            if(nr>=0 && nr<3 && nc>=0 && nc<3){
                State nextState=current->state;
                swap(nextState[r][c], nextState[nr][nc]);
                if(!closedList.count(nextState)) openList.push(new Node(nextState, current->g+1, moves[i], current));
            }
        }
    }
}
State inputState(string name){
    State s(3,vector<int>(3));
    cout << "Enter " << name << " State (3x3 grid, use 0 for blank):" << endl;
    for(int i=0;i<3;i++){
        for(int j=0;j<3;j++){
            cin >> s[i][j];
        }
    }
    return s;
}
int main() {
    State startState = inputState("Start");
    State goalState = inputState("Goal");

    cout << "\nChecking solvability..." << endl;
    if (isSolvable(startState, goalState)) {
        solvePuzzle(startState, goalState);
    } else {
        cout << "IMPOSSIBLE: The parity of inversions does not match." << endl;
        cout << "This start state cannot reach this goal state." << endl;
    }
    return 0;
}
